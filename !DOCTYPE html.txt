        function cambiarPestana(pestana) {
            const btnGrupos = document.getElementById('btn-grupos');
            const btnEliminatorias = document.getElementById('btn-eliminatorias');
            const secGrupos = document.getElementById('seccion-grupos');
            const secEliminatorias = document.getElementById('seccion-eliminatorias');

            if (pestana === 'grupos') {
                btnGrupos.className = "px-6 py-3 rounded-xl font-bold uppercase text-sm tracking-wider transition neon-border bg-[#131b2e] text-[#ccff00]";
                btnEliminatorias.className = "px-6 py-3 rounded-xl font-bold uppercase text-sm tracking-wider transition bg-[#131b2e] text-gray-400 border border-gray-800 hover:border-gray-600";
                secGrupos.classList.remove('hidden');
                secEliminatorias.classList.add('hidden');
            } else {
                btnEliminatorias.className = "px-6 py-3 rounded-xl font-bold uppercase text-sm tracking-wider transition neon-border bg-[#131b2e] text-[#ccff00]";
                btnGrupos.className = "px-6 py-3 rounded-xl font-bold uppercase text-sm tracking-wider transition bg-[#131b2e] text-gray-400 border border-gray-800 hover:border-gray-600";
                secEliminatorias.classList.remove('hidden');
                secGrupos.classList.add('hidden');
            }
        }
